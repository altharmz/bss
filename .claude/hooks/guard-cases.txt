# Regression cases for guard.sh, run by .claude/hooks/test-guard.sh (and by the
# pre-commit hook whenever .claude/ changes). One case per line:
#   allow <command>      expected exit 0
#   block <command>      expected exit 2
#   allow-ml / block-ml  the same, with every literal \n in the command turned into a newline
# Paths are relative to the project root; absolute ones use /home/petrg/projects/bss.

# --- the VPN on the VPS (raw text, quotes included) ---
block ssh vps-test 'vpncmd /server localhost'
block docker exec vpnserver ls
block echo softether
block ssh vps-test 'ss -tlnp | grep 443'
block docker run -p 443:443 nginx

# --- VPN ports: blocked in a port/network context, not as bare numbers elsewhere ---
allow docker compose run --rm app pytest -p no:xdist tests/test_api_conventions.py >/tmp/x/neg-500-unfixed.log 2>&1
allow docker compose run --rm app pytest -k "status_code == 500"
allow curl -s -o /dev/null -w '%{http_code}' http://localhost:8080/healthz
allow docker compose exec app curl -s -o /dev/null -w '%{http_code}' http://localhost:8080/boom | grep -qx 500
allow docker logs bss-app-1 | grep ' 500 '
allow docker compose run --rm app pytest tests/test_x.py::test_retry_after_4500_ms
block ufw allow 500
block docker run -p 4500:4500 nginx
block ssh vps-test 'ss -ltnp | grep 1194'
block nc -z vpn2811.duckdns.org 992
block ss -ltnp | grep 1194
block docker run --publish=127.0.0.1:5555:5555 nginx
block sh -c 'docker run -p 443:443 nginx'
block curl -sk https://vpn2811.duckdns.org:443/
block sudo iptables -A INPUT -p udp --dport 500 -j ACCEPT
block ssh -N -L 992:127.0.0.1:992 vps-test
block docker compose run --rm app python -m http.server --port 443
block-ml cat <<'EOF'\nufw allow 500\nEOF
block-ml cat <<'EOF'\ncurl -sk https://vpn2811.duckdns.org:443/\nEOF
block-ml cat <<'EOF'\ndocker run -p 4500:4500 nginx\nEOF
# fail-closed on purpose: a heredoc is untokenisable, so the broad raw-text rule applies
# even to a commit message (CLAUDE.md: commit messages go through -F <file>)
block-ml git commit -F - <<'EOF'\nguard: ports 443 and 500 next to ssh and docker\nEOF
